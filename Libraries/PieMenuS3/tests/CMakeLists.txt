cmake_minimum_required(VERSION 3.20)
project(piemenu_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ARDUINO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_CONF_SKIP ON CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_compile_definitions(LV_COLOR_FORMAT_DEFAULT=LV_COLOR_FORMAT_RGB565 LV_MEM_SIZE=98304 LV_FONT_MONTSERRAT_28=1 LV_BUILD_EXAMPLES=0 LV_BUILD_DEMOS=0)
add_subdirectory("${ARDUINO_ROOT}/Libraries/lvgl" lvgl)
add_executable(piemenu_tests test_piemenu.cpp ../lv_piemenu.cpp)
target_link_libraries(piemenu_tests PRIVATE lvgl)
target_include_directories(piemenu_tests PRIVATE .. "${ARDUINO_ROOT}/Libraries/lvgl")
enable_testing()
add_test(NAME piemenu COMMAND piemenu_tests)
